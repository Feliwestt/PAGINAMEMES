<?php
/* admin/crear_admin.php
   Permite a un administrador crear otro usuario administrador
-----------------------------------------------------------------*/
require '_auth_admin.php';                 // protege la página
require __DIR__.'includes/conexion.php';  // $conexion = mysqli_connect(...)

$mensaje = '';      // feedback para la vista
$errores = [];

if ($_SERVER['REQUEST_METHOD'] === 'POST') {

    // 1. Recoge y valida campos
    $username = trim($_POST['username'] ?? '');
    $password = $_POST['password'] ?? '';

    if ($username === '') {
        $errores[] = 'El nombre de usuario es obligatorio';
    }
    if ($password === '') {
        $errores[] = 'La contraseña no puede estar vacía';
    }

    if (!$errores) {
        // 2. Verifica que el usuario no exista
        $check = $conexion->prepare("SELECT 1 FROM admins WHERE usuario = ? LIMIT 1");
        $check->bind_param('s', $username);
        $check->execute();
        $check->store_result();

        if ($check->num_rows) {
            $errores[] = "El usuario <strong>$username</strong> ya existe";
        } else {
            // 3. Inserta nuevo admin con la contraseña hasheada
            $hash = password_hash($password, PASSWORD_DEFAULT);
            $ins  = $conexion->prepare(
                "INSERT INTO admins (usuario, contra_hash) VALUES (?,?)"
            );
            $ins->bind_param('ss', $username, $hash);

            if ($ins->execute()) {
                $mensaje = "✅ Admin <strong>$username</strong> creado correctamente.";
            } else {
                $errores[] = 'Error al insertar: '.$ins->error;
            }
            $ins->close();
        }
        $check->close();
    }
}
